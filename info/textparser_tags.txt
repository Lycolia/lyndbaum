#（注意）
#　・先頭 '#' はコメント。空行は無視。
#　・alias は ">linkname"。１階層しかたどれない。
#　・url には「,」を含めない。
#  ・文字コード "ASCII" や 小文字指定 はコード変換しない。
#  ・引数の数が限定されていれば、それ以上の表記は「:」で連結する。
#------------------------------------------------------------------------------
# ■サイト固有のタグ設定ファイルのロード
#------------------------------------------------------------------------------
load textparser_site_tags.txt

#------------------------------------------------------------------------------
# adiary用プラグイン, adiary設定
#------------------------------------------------------------------------------
# plugin adiary
link = >adiary:link
this = >adiary:this
key  = >adiary:key
tm   = >adiary:tm
day  = >adiary:day
tag  = >adiary:tag
adiary:link= 記事指定
adiary:this= 記事内リンク
adiary:key = 記事指定
adiary:tm  = 記事時刻指定
adiary:day = 日付指定
adiary:tag = タグ検索

search = 検索,         UTF-8,  ${myself}?all=1&q=
title  = タイトル検索, UTF-8,  ${myself}?q=

category = >adiary:tag
q  = >search
qt = >title

#------------------------------------------------------------------------------
# 色々
#------------------------------------------------------------------------------
# youtube は Tag_url_filter.pm（サムネイルを出すため）
niconico    = ニコニコ動画, ASCII, 1, <module name="nico" vid="$1">

#------------------------------------------------------------------------------
# 画像プラグイン／メディア管理
#------------------------------------------------------------------------------
# plugin image
# plugin exif

# ただの画像
img        = 画像, image

# メディア管理の画像
image:L      = メディア管理の画像, image, 2,  ${image}$1$2
# サムネイル。AVIFのサムネイル（.avif）があれば、Tag_image.pmがそちらにする
image:S      = メディア管理の画像, image, 2,  ${image}$1.thumbnail/$2.jpg
image:S#link = メディア管理の画像, image, 2,  ${image}$1$2
# 画像へのテキストリンク（#textlinkがあると、画像を出さずにリンクの文字を出す。Tag_image.pm）
image:T          = 画像へのリンク, image, 2,  ${image}$1$2
image:T#textlink = 画像へのリンク, image, 2,  ${image}$1$2
# 360度写真（sp_view360プラグインで見回せる。tag-image-360のクラスが付く）
image:360    = 360度写真,    image, 2,  ${image}$1$2

# 通常ファイル
file        = ファイル, ASCII, 3,  ${image}$2$3
file#5      = ファイル, ASCII, 4,  ${image}$3$4

# exif
exif#link = 画像へのリンク, ASCII, 2, ${image}$1$2

#------------------------------------------------------------------------------
# Audio/Video
#------------------------------------------------------------------------------
file:wav  = >audio
file:wave = >audio
file:ogg  = >audio
file:oga  = >audio
file:mp3  = >audio
file:aac  = >audio
file:m4a  = >audio

file:ogv  = >video
file:webm = >video
file:mp4  = >video
file:m4v  = >video

#------------------------------------------------------------------------------
# その他
#------------------------------------------------------------------------------
# inline comment
comment = comment,, 1, <!--#1-->
# inline class
class = class, ASCII, 2, <span class="$1">#2</span>

# ルビ
ruby = ルビ, ASCII, 2, <ruby>#1<rp>(</rp><rt>#2</rt><rp>)</rp></ruby>

# tel
tel   = <a href="tel:$0">#0</a>
sms   = <a href="sms:$0">#0</a>

# 内部変数置換
myself  = text:${myself}
myself2 = text:${myself2}
public  = text:${public}
pubdist = text:${pubdist}

#------------------------------------------------------------------------------
# HTMLタグへの置換
#------------------------------------------------------------------------------
em	= html:em

del	= html:del
ins	= html:ins
u	= html:span.underline
h	= html:span.hidden
hidden	= html:span.hidden
pre	= html:span.pre
sup	= html:sup
sub	= html:sub
clear   = html:br.clear
wbr     = html:wbr
code	= html:code
quote	= html:q

#------------------------------------------------------------------------------
# テキスト色
#------------------------------------------------------------------------------
color = テキスト色, ASCII, 2, <span style="color:#1;">#2</span>

#------------------------------------------------------------------------------
# 麻雀記法
#------------------------------------------------------------------------------
# plugin mahjong
mj:img = 麻雀画像, ASCII, 0, ${pubdist}mahjong/

#------------------------------------------------------------------------------
# 文字サイズ

xx-large = html:span.xx-large
x-large  = html:span.x-large
large    = html:span.large
small    = html:span.small
x-small  = html:span.x-small
xx-small = html:span.xx-small
